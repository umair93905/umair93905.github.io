<script>
  document.addEventListener('DOMContentLoaded', function () {
    var projectCards = Array.prototype.slice.call(document.querySelectorAll('[data-project-card]'));

    function setDetailFocus(details, shouldAllowFocus) {
      var focusableItems = details.querySelectorAll('a, button, input, select, textarea, [tabindex]');

      Array.prototype.forEach.call(focusableItems, function (item) {
        if (shouldAllowFocus) {
          item.removeAttribute('tabindex');
          return;
        }

        item.setAttribute('tabindex', '-1');
      });
    }

    function setExpanded(card, shouldExpand) {
      var toggle = card.querySelector('.project-toggle');
      var details = card.querySelector('.project-details');

      if (!toggle || !details) {
        return;
      }

      if (shouldExpand) {
        projectCards.forEach(function (otherCard) {
          if (otherCard !== card) {
            setExpanded(otherCard, false);
          }
        });

        card.classList.add('is-expanded');
        toggle.setAttribute('aria-expanded', 'true');
        details.setAttribute('aria-hidden', 'false');
        setDetailFocus(details, true);
        details.style.maxHeight = details.scrollHeight + 'px';
        return;
      }

      card.classList.remove('is-expanded');
      toggle.setAttribute('aria-expanded', 'false');
      details.setAttribute('aria-hidden', 'true');
      setDetailFocus(details, false);
      details.style.maxHeight = '0px';
    }

    function refreshExpandedHeights() {
      projectCards.forEach(function (card) {
        var details = card.querySelector('.project-details');

        if (details && card.classList.contains('is-expanded')) {
          details.style.maxHeight = details.scrollHeight + 'px';
        }
      });
    }

    projectCards.forEach(function (card) {
      var details = card.querySelector('.project-details');

      if (details) {
        setDetailFocus(details, false);
        details.style.maxHeight = '0px';
      }

      card.addEventListener('click', function (event) {
        if (event.target.closest('a')) {
          return;
        }

        setExpanded(card, !card.classList.contains('is-expanded'));
      });
    });

    window.addEventListener('resize', refreshExpandedHeights);
    window.addEventListener('load', refreshExpandedHeights);
  });
</script>
